# 交互式终端（psql）

# 1. 最简登录方式

psql
1

前提条件：当前系统用户与 PostgreSQL 数据库用户同名（如已 sudo -i -u postgres 切换），且数据库为本机默认端口 5432。

# 2. 不同场景下的参数补充

场景 命令 说明
已切换 postgres 用户 psql 零参数，直接以 postgres 身份进入同名数据库
普通用户指定数据库用户 psql -U postgres 最少需带 -U 指定用户名
指定用户 + 指定数据库 psql -U postgres -d testdb 同时指定用户和数据库名
远程连接 psql -h 192.168.1.10 -U postgres -d testdb 需额外指定主机地址 -h

# 3. 常用参数速查

参数 全称 说明
-h --host 服务器地址（默认本机）
-p --port 端口号（默认 5432）
-U --username 数据库用户名
-d --dbname 数据库名
-W --password 强制提示输入密码

# 4. 完整写法对照

# 完整写法（远程连接）
psql -h <服务器地址> -p <端口号> -d <数据库名> -U <用户名>

# 本地默认连接（当前系统用户同名数据库）
psql

# 指定用户连接默认数据库
psql -U postgres

# 指定用户 + 指定数据库
psql -U postgres -d testdb

# 5. 常用元命令速查

命令 说明
\l 列出所有数据库
\c dbname 切换到指定数据库
\dn 列出所有模式
\d 列出当前数据库所有表
\d tablename 查看表结构详情
\dt 只列出普通表
\di 列出索引
\du 列出所有用户/角色
\q 退出 psql

提示：\ 开头的命令是 psql 客户端元命令，不是 SQL 语句，不需要以分号结尾；SQL 语句必须以 ; 结尾。

# 多用户权限管理实战（连贯示例）

命名规范说明

本实战采用 biz_{场景}_{权限} 统一命名，biz_ 为业务前缀，{场景} 标识用途（owner/app/auto/supp），{权限} 标识级别（admin/rw/ro）。角色模板统一使用 biz_role_{权限}（NOLOGIN），实现“用户与权限解耦”。

权限梯度与职责

  • admin 级（DDL+DML）：biz_owner_admin 为 Schema 唯一归属者；biz_auto_admin 供 CI/Release 工具使用，通过 INHERIT + SET ROLE 临时获得 Owner 权限执行建表/改表，日常操作安全可控。
  • rw 级（仅 DML）：biz_role_rw 为读写模板。biz_app_rw（Java 应用连接）、biz_supp_rw（Support 手动数据修复）均继承此角色，具备 CRUD 能力但无表结构变更权。
  • ro 级（只读）：biz_role_ro 为只读模板。biz_supp_ro 供 Support 团队排查问题，仅限 SELECT。

核心优势

  1. 一眼懂：后缀即权限（admin/rw/ro），中缀即场景，零学习成本。
  2. 自动继承：通过 ALTER DEFAULT PRIVILEGES 确保新表自动对 rw/ro 角色授权，免维护。
  3. 最小权限：应用与运维账号不持有 DDL 权限，降低误删表风险；自动化工具需显式 SET ROLE 才具建表权。

一眼看懂的命名树：

biz_owner_admin  (Schema Owner, DDL+DML)
  ▲
  │  INHERIT + SET ROLE
biz_auto_admin   (CI/Release, DDL+DML)

biz_role_rw  (NOLOGIN, 权限模板)
  ├──► biz_app_rw   (Java应用)
  └──► biz_supp_rw  (Support运维)

biz_role_ro    (NOLOGIN, 权限模板)
  └──► biz_supp_ro    (Support只读)

完整建库脚本：

-- ============================================================
-- 1. 初始化连接（用超级用户 postgres）
-- ============================================================
psql -U postgres -d postgres

-- 确认当前用户和当前 schema
SELECT current_user, current_schema();


-- ============================================================
-- 2. 创建 biz_owner_admin —— Schema Owner (DDL+DML)
--   LOGIN        : 允许登录
--   CREATEDB     : 允许创建数据库
--   CREATEROLE   : 允许创建/管理角色
-- ============================================================
CREATE USER biz_owner_admin WITH PASSWORD '123456'
  LOGIN
  CREATEDB
  CREATEROLE;


-- ============================================================
-- 3. 切换到 biz_owner_admin，创建数据库和 schema
-- ============================================================
psql -U biz_owner_admin -d postgres

-- 创建业务数据库
CREATE DATABASE bizdb;

-- 连接到业务库
\c bizdb

-- 创建业务 schema
CREATE SCHEMA bizschema;

-- 显式授予 schema 权限：授予 biz_owner_admin 在该 schema 中建表和查询的权限
-- CREATE : 允许新建表/视图/函数等
-- USAGE  : 允许访问 schema 中的对象
GRANT CREATE, USAGE ON SCHEMA bizschema TO biz_owner_admin;

-- 设置 biz_owner_admin 的默认 search_path
-- 效果：建表/查表时优先使用 bizschema
-- ⚠️ 需要重新连接才会生效
ALTER ROLE biz_owner_admin SET search_path TO bizschema, public;


-- ============================================================
-- 4. 重新以 biz_owner_admin 登录 bizdb，验证 schema 默认行为
-- ============================================================
\q
psql -U biz_owner_admin -d bizdb

-- ----------------------------
-- 建表 1
-- ----------------------------
CREATE TABLE test_table_1 (
    col_1 INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    col_2 VARCHAR(50) NOT NULL,
    col_3 TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- CRUD 验证 —— test_table_1
SELECT * FROM test_table_1;

INSERT INTO test_table_1 (col_2) VALUES ('value');
UPDATE test_table_1 SET col_2 = 'value_updated' WHERE col_2 = 'value';
DELETE FROM test_table_1 WHERE col_2 = 'value_updated';

-- DDL 验证 —— test_table_1
ALTER TABLE test_table_1 ADD COLUMN col_4 VARCHAR(50);
ALTER TABLE test_table_1 DROP COLUMN col_4;


-- ============================================================
-- 5. 创建"自动化/子用户" biz_auto_admin
--   INHERIT : 自动继承所属角色的权限
-- ============================================================
CREATE USER biz_auto_admin WITH PASSWORD '123456'
  LOGIN
  INHERIT;

-- 设置默认 schema（同样需要重连生效）
ALTER ROLE biz_auto_admin SET search_path TO bizschema, public;

-- 显式授予 schema 权限：允许该用户在 schema 中创建对象
GRANT CREATE, USAGE ON SCHEMA bizschema TO biz_auto_admin;

\q
psql -U postgres -d postgres

-- ============================================================
-- 6. 建立角色继承关系，让 CI 用户成为 schema owner 的成员
--   biz_biz_auto_admin 继承 biz_owner_admin 的权限
--   WITH INHERIT TRUE : 自动享有 biz_owner_admin 的权限
--   WITH SET TRUE     : 允许 biz_biz_auto_admin 用 SET ROLE 切换为 biz_owner_admin
-- ============================================================
-- REVOKE biz_auto_admin FROM biz_owner_admin;
GRANT biz_owner_admin TO biz_auto_admin
  WITH INHERIT TRUE, SET TRUE;

\q
psql -U biz_auto_admin -d bizdb

-- ============================================================
-- 7. 演示：以 biz_auto_admin 身份，通过 SET ROLE 创建表
--   这样表的 owner 是 biz_owner_admin，而不是 biz_auto_admin
-- ============================================================
SET ROLE biz_owner_admin;

-- ----------------------------
-- 建表 2
-- ----------------------------
CREATE TABLE test_table_2 (
    col_1 INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    col_2 VARCHAR(50) NOT NULL,
    col_3 TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 切回原角色
RESET ROLE;

-- CRUD 验证 —— test_table_2
SELECT * FROM test_table_2;

INSERT INTO test_table_2 (col_2) VALUES ('value');
UPDATE test_table_2 SET col_2 = 'value_updated' WHERE col_2 = 'value';
DELETE FROM test_table_2 WHERE col_2 = 'value_updated';

-- DDL 验证 —— test_table_2
ALTER TABLE test_table_2 ADD COLUMN col_4 VARCHAR(50);
ALTER TABLE test_table_2 DROP COLUMN col_4;
